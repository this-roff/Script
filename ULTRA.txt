local Utils = {}

function Utils.GetClosestPlayer(fov)
    fov = fov or 9999
    local closest, distance = nil, math.huge
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local root = plr.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
                if dist < distance and dist < fov then
                    closest, distance = plr, dist
                end
            end
        end
    end
    return closest
end

function Utils.GetPlayersInFOV(fov)
    fov = fov or 9999
    local players = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local root = plr.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
                if dist < fov then
                    table.insert(players, {player = plr, distance = dist})
                end
            end
        end
    end
    table.sort(players, function(a, b) return a.distance < b.distance end)
    return players
end

function Utils.GetPart(plr, partName)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild(partName) or plr.Character:FindFirstChild("HumanoidRootPart")
end

function Utils.GetHumanoid(plr)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild("Humanoid")
end

function Utils.IsAlive(plr)
    local hum = Utils.GetHumanoid(plr)
    return hum and hum.Health > 0
end

function Utils.GetDistance(plr)
    if not plr or not plr.Character then return math.huge end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return math.huge end
    return (Camera.CFrame.Position - root.Position).magnitude
end

function Utils.IsTeamMate(plr)
    if not Config.AimTeamCheck then return false end
    return plr.Team == LocalPlayer.Team
end

function Utils.IsVisible(plr)
    if not plr or not plr.Character then return false end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local origin = Camera.CFrame.Position
    local direction = (root.Position - origin).unit * 500
    local ray = Ray.new(origin, direction)
    local hit = Workspace:FindPartOnRay(ray, LocalPlayer.Character)
    if hit then
        local hitPlr = hit.Parent and hit.Parent.Parent and hit.Parent.Parent:FindFirstChild("Humanoid")
        if hitPlr then
            return hitPlr == Utils.GetHumanoid(plr)
        end
    end
    return false
end

function Utils.Lerp(a, b, t)
    return a + (b - a) * t
end

function Utils.Clamp(value, min, max)
    return math.max(min, math.min(max, value))
end

local Aim = {
    Active = false,
    LockedTarget = nil,
    Smoothness = 0.25,
    FOV = 150,
    TargetPart = "Head",
    TeamCheck = true,
    Prediction = false,
    PredictionAmount = 0.2,
    SilentAim = false,
    SilentActive = false,
}

function Aim:FindTarget()
    local target = Utils.GetClosestPlayer(self.FOV)
    if target and Utils.IsAlive(target) then
        if self.TeamCheck and Utils.IsTeamMate(target) then return nil end
        return target
    end
    return nil
end

function Aim:GetTargetPosition(plr)
    local part = Utils.GetPart(plr, self.TargetPart)
    if not part then return nil end
    local pos = part.Position
    if self.Prediction and plr.Character then
        local root = plr.Character:FindFirstChild("HumanoidRootPart")
        if root then
            pos = pos + root.Velocity * self.PredictionAmount
        end
    end
    return pos
end

function Aim:Lock()
    while self.Active do
        local target = self:FindTarget()
        if target then
            self.LockedTarget = target
            local pos = self:GetTargetPosition(target)
            if pos then
                local lookAt = CFrame.lookAt(Camera.CFrame.Position, pos)
                Camera.CFrame = Camera.CFrame:Lerp(lookAt, self.Smoothness)
            end
        else
            self.LockedTarget = nil
        end
        wait(0.01)
    end
end

function Aim:SilentAim()
    while self.SilentActive do
        local target = self:FindTarget()
        if target then
            local pos = self:GetTargetPosition(target)
            if pos then
                Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, pos)
            end
        end
        wait(0.01)
    end
end

function Aim:Toggle()
    self.Active = not self.Active
    if self.Active then
        task.spawn(function() self:Lock() end)
        print("Aim Lock: ON")
    else
        self.LockedTarget = nil
        print("Aim Lock: OFF")
    end
end

function Aim:ToggleSilent()
    self.SilentActive = not self.SilentActive
    if self.SilentActive then
        task.spawn(function() self:SilentAim() end)
        print("Silent Aim: ON")

    end
end

function Trigger:FindShootRemote()
    if self.ShootRemote then return self.ShootRemote end
    for _, remote in pairs(ReplicatedStorage:GetChildren()) do
        if remote:IsA("RemoteEvent") and remote.Name:match("[Ss]hoot") then
            self.ShootRemote = remote
            return remote
        end
    end
    for _, plr in pairs(Players:GetPlayers()) do
        if plr.Character then
            for _, part in pairs(plr.Character:GetChildren()) do
                if part:IsA("Tool") and part:FindFirstChild("Shoot") then
                    local remote = part:FindFirstChild("Shoot")
                    if remote and remote:IsA("RemoteEvent") then
                        self.ShootRemote = remote
                        return remote
                    end
                end
            end
        end
    end
    return nil
end

function Trigger:Shoot()
    local remote = self:FindShootRemote()
    if remote then
        remote:FireServer()
    else
        VirtualUser:CaptureController()
        VirtualUser:ClickButton(Vector2.new(0, 0))
    end
end

function Trigger:CheckTarget(plr)
    if not plr or not plr.Character then return false end
    if self.TeamCheck and Utils.IsTeamMate(plr) then return false end
    if not Utils.IsAlive(plr) then return false end
    
    local part = self.OnlyHead and plr.Character:FindFirstChild("Head") or plr.Character:FindFirstChild("HumanoidRootPart")
    if not part then return false end
    
    local pos, onScreen = Camera:WorldToScreenPoint(part.Position)
    if not onScreen then return false end
    
    local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
    return dist < self.FOV
end

function Trigger:Run()
    while self.Active do
        for _, plr in pairs(Players:GetPlayers()) do
            if self:CheckTarget(plr) then
                self:Shoot()
                break
            end
        end
        wait(self.Delay)
    end
end

function ESP:CreateESP(plr)
    if not plr.Character then return end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    local color = self:GetPlayerColor(plr)
    local data = {plr = plr}
    
    -- Box ESP
    if self.Box then
        local box = Instance.new("BoxHandleAdornment")
        box.Size = Vector3.new(3, 5, 1)
        box.Adornee = root
        box.Color3 = color
        box.Thickness = self.BoxThickness
        box.AlwaysOnTop = true
        box.Visible = true
        box.Parent = root
        data.box = box
    end
    
    -- Name ESP
    if self.Name then
        local gui = Instance.new("BillboardGui")
        gui.Size = UDim2.new(0, 150, 0, 25)
        gui.Adornee = root
        gui.Parent = root
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = plr.Name
        label.TextColor3 = self.NameColor
        label.TextSize = self.NameSize
        label.TextScaled = true
        label.Font = Enum.Font.GothamBold
        label.Parent = gui
        data.name = gui
    end
    
    -- Health ESP
    if self.Health or self.HealthBar then
        local gui = Instance.new("BillboardGui")
        gui.Size = UDim2.new(0, 50, 0, 5)
        gui.Adornee = root
        gui.Position = UDim2.new(0, -25, 0, -3)
        gui.Parent = root
        local bar = Instance.new("Frame")
        bar.Size = UDim2.new(1, 0, 1, 0)
        bar.BackgroundColor3 = self.HealthColor
        bar.BackgroundTransparency = 0.3
        bar.Parent = gui
        data.health = {gui = gui, bar = bar}
    end
    
    -- Distance ESP
    if self.Distance then
        local gui = Instance.new("BillboardGui")
        gui.Size = UDim2.new(0, 60, 0, 15)
        gui.Adornee = root
        gui.Position = UDim2.new(0, -30, 0, 25)
        gui.Parent = root
        local label = Instance.new("TextLabel")
        label.Size = UDim2.new(1, 0, 1, 0)
        label.BackgroundTransparency = 1
        label.Text = "0m"
        label.TextColor3 = self.DistanceColor
        label.TextScaled = true
        label.Font = Enum.Font.Gotham
        label.Parent = gui
        data.dist = gui
    end
    
    table.insert(self.Objects, data)
end

function ESP:UpdateESP()
    while self.Active do
        -- Cleanup
        for i, data in pairs(self.Objects) do
            if not data.plr or not data.plr.Character then
                if data.box then data.box:Destroy() end
                if data.name then data.name:Destroy() end
                if data.health then data.health.gui:Destroy() end
                if data.dist then data.dist:Destroy() end
                table.remove(self.Objects, i)
            else
                -- Update Health
                if data.health then
                    local hum = Utils.GetHumanoid(data.plr)
                    if hum then
                        local pct = hum.Health / hum.MaxHealth
                        data.health.bar.Size = UDim2.new(pct, 0, 1, 0)
                        data.health.bar.BackgroundColor3 = pct > 0.5 and Color3.fromRGB(0, 255, 0) or 
                                                             (pct > 0.25 and Color3.fromRGB(255, 255, 0) or Color3.fromRGB(255, 0, 0))
                    end
                end
                -- Update Distance
                if data.dist then
                    local dist = Utils.GetDistance(data.plr)
                    data.dist.TextLabel.Text = math.floor(dist) .. "m"
                end
                -- Update Color
                local color = self:GetPlayerColor(data.plr)
                if data.box then data.box.Color3 = color end
            end
        end
        
        -- Create new ESP
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local exists = false
                for _, data in pairs(self.Objects) do
                    if data.plr == plr then exists = true; break end
                end
                if not exists then self:CreateESP(plr) end
            end
        end
        wait(0.1)
    end
end

function ESP:Toggle()
    self.Active = not self.Active
    if self.Active then
        task.spawn(function() self:UpdateESP() end)
        print("ESP: ON")
    else
        for _, data in pairs(self.Objects) do
            if data.box then data.box:Destroy() end
            if data.name then data.name:Destroy() end
            if data.health then data.health.gui:Destroy() end
            if data.dist then data.dist:Destroy() end
        end
        self.Objects = {}
        print("ESP: OFF")
    end
end

function Skeleton:Create(plr)
    if not plr.Character then return end
    local parts = {"Head", "Torso", "LeftArm", "RightArm", "LeftLeg", "RightLeg"}
    local connections = {{"Head", "Torso"}, {"Torso", "LeftArm"}, {"Torso", "RightArm"}, {"Torso", "LeftLeg"}, {"Torso", "RightLeg"}}
    local color = self:GetColor(plr)
    local lines = {}
    for _, conn in pairs(connections) do
        local a = plr.Character:FindFirstChild(conn[1])
        local b = plr.Character:FindFirstChild(conn[2])
        if a and b then
            local line = Instance.new("LineHandleAdornment")
            line.Length = (a.Position - b.Position).magnitude
            line.Thickness = self.Thickness
            line.Color3 = color
            line.Adornee = a
            line.AlwaysOnTop = true
            line.Parent = a
            table.insert(lines, line)
        end
    end
    return lines
end

function Skeleton:Update()
    while self.Active do
        for i, data in pairs(self.Objects) do
            if not data.plr or not data.plr.Character then
                for _, line in pairs(data.lines) do
                    line:Destroy()
                end
                table.remove(self.Objects, i)
            else
                local color = self:GetColor(data.plr)
                for _, line in pairs(data.lines) do
                    line.Color3 = color
                end
            end
        end
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local exists = false
                for _, data in pairs(self.Objects) do
                    if data.plr == plr then exists = true; break end

                    if lines then
                        table.insert(self.Objects, {plr = plr, lines = lines})

                end
            end
        end
        wait(0.5)

local AutoFarm = {
    Active = false,
    Mode = "Kill",
    Delay = 0.1,
    FOV = 200,
    TeamCheck = true,
    Collect = false,
    CollectRadius = 50,
    Heal = false,
    HealThreshold = 30,
    Reload = false,
    ReloadDelay = 2,
    Stats = {Kills = 0, Clicks = 0, Collects = 0, StartTime = 0},
}

function AutoFarm:FindTarget()
    return Utils.GetClosestPlayer(self.FOV)
end

function AutoFarm:Shoot(target)
    if not target then return end
    local remote = Trigger.ShootRemote or Trigger:FindShootRemote()
    if remote then
        remote:FireServer()
        self.Stats.Clicks = self.Stats.Clicks + 1
    else
        VirtualUser:CaptureController()
        VirtualUser:ClickButton(Vector2.new(0, 0))
        self.Stats.Clicks = self.Stats.Clicks + 1
    end
    if target and Utils.IsAlive(target) then
        local hum = Utils.GetHumanoid(target)
        if hum and hum.Health <= 0 then
            self.Stats.Kills = self.Stats.Kills + 1
        end
    end
end

function AutoFarm:HealPlayer()
    if not LocalPlayer.Character then return end
    local hum = Utils.GetHumanoid(LocalPlayer)
    if hum and hum.Health < self.HealThreshold then
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        if backpack then
            for _, item in pairs(backpack:GetChildren()) do
                if item:IsA("Tool") and item.Name:match("[Hh]eal") then
                    item.Parent = LocalPlayer.Character
                    item:Activate()
                    wait(0.5)
                    item.Parent = backpack
                    break
                end
            end
        end
    end
end

function AutoFarm:CollectItems()
    if not self.Collect then return end
    for _, item in pairs(Workspace:GetChildren()) do
        if item:IsA("BasePart") and item.Name:match("[Cc]ollect") then
            local dist = (item.Position - LocalPlayer.Character.HumanoidRootPart.Position).magnitude
            if dist < self.CollectRadius then
                local remote = ReplicatedStorage:FindFirstChild("Collect")
                if remote then
                    remote:FireServer(item)
                    self.Stats.Collects = self.Stats.Collects + 1
                end
            end
        end
    end
end

function AutoFarm:ReloadWeapon()
    if not LocalPlayer.Character then return end
    local tool = LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("Reload") then
        tool:FindFirstChild("Reload"):FireServer()
    end
end

function AutoFarm:Run()
    while self.Active do
        if self.Heal then self:HealPlayer() end
        if self.Reload then self:ReloadWeapon() end
        if self.Collect then self:CollectItems() end
        
        if self.Mode == "Kill" or self.Mode == "Click" then
            local target = self:FindTarget()
            if target then
                self:Shoot(target)
            end
        end
        wait(self.Delay)
    end
end

function AutoFarm:Toggle()
    self.Active = not self.Active
    if self.Active then
        self.Stats.StartTime = os.time()
        task.spawn(function() self:Run() end)
        print("Auto-Farm: ON")
    else
        print("Auto-Farm: OFF")
        print("📊 Stats: Kills=" .. self.Stats.Kills .. " Clicks=" .. self.Stats.Clicks .. " Collects=" .. self.Stats.Collects)
    end
end

function AutoFarm:SetMode(mode)
    if mode == "Kill" or mode == "Click" or mode == "Collect" then
        self.Mode = mode
        print("Auto-Farm Mode:", mode)
    end
end

function ROFFHub:SetupDrag(frame)

        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

        if dragging and dragInput and startPos then
            local delta = dragInput.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale + delta.X / GuiService.ScreenSize.X,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale + delta.Y / GuiService.ScreenSize.Y,

        end
        wait(0.1)
    end
end

    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and Utils.IsAlive(plr) then
            if self.TeamCheck and Utils.IsTeamMate(plr) then continue end
            local distance = Utils.GetDistance(plr)
            if distance > self.MaxDistance then continue end
            if self.VisibleCheck and not Utils.IsVisible(plr) then continue end
            
            local root = Utils.GetPart(plr, self.TargetPart)
            if not root then continue end
            local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
            if not onScreen then continue end
            
            local fovDist = (Vector2.new(pos.X, pos.Y) - center).magnitude
            if fovDist > self.FOV then continue end
            
            local hum = Utils.GetHumanoid(plr)
            if hum and hum.Health <= self.MinHealth then continue end
            
            local score = 0
            if self.Mode == "FOV" then
                score = 1000 - fovDist
            elseif self.Mode == "Distance" then
                score = 1000 - distance
            elseif self.Mode == "Health" then
                score = hum and hum.Health or 0
            elseif self.Mode == "Crosshair" then
                score = 1000 - fovDist
                score = score + (hum and hum.Health or 0) * 0.1
            end
            
            -- Bonus jika target sudah pernah dilock
            if self.CurrentTarget == plr then
                score = score + 500
            end
            
            table.insert(candidates, {player = plr, score = score, distance = distance, fov = fovDist})
        end
    end
    
    table.sort(candidates, function(a, b) return a.score > b.score end)
    if #candidates > 0 then
        return candidates[1]
    end
    return nil
end

function AdvancedAim:GetTargetPosition(plr)
    local part = Utils.GetPart(plr, self.TargetPart)
    if not part then return nil end
    local pos = part.Position
    
    if self.Prediction and plr.Character then
        local root = plr.Character:FindFirstChild("HumanoidRootPart")
        if root then
            local velocity = root.Velocity
            local distance = Utils.GetDistance(plr)
            local travelTime = distance / 300 -- Asumsi kecepatan peluru 300 stud/s
            pos = pos + velocity * travelTime * self.PredictionAmount
        end
    end
    return pos
end

function TargetPriority:SortTargets(targets)

        table.sort(targets, function(a, b)
            return Utils.GetDistance(a) < Utils.GetDistance(b)
        end)
    elseif self.Mode == "Health" then
        table.sort(targets, function(a, b)
            local ha = Utils.GetHumanoid(a)
            local hb = Utils.GetHumanoid(b)
            return (ha and ha.Health or 0) < (hb and hb.Health or 0)
        end)
    elseif self.Mode == "FOV" then
        local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        table.sort(targets, function(a, b)
            local ra = Utils.GetPart(a, "HumanoidRootPart")
            local rb = Utils.GetPart(b, "HumanoidRootPart")
            if ra and rb then
                local pa, _ = Camera:WorldToScreenPoint(ra.Position)
                local pb, _ = Camera:WorldToScreenPoint(rb.Position)
                return (Vector2.new(pa.X, pa.Y) - center).magnitude < (Vector2.new(pb.X, pb.Y) - center).magnitude
            end
            return false
        end)
    elseif self.Mode == "Random" then
        for i = #targets, 2, -1 do
            local j = math.random(i)
            targets[i], targets[j] = targets[j], targets[i]
        end
    end
    return targets
end

    local targets = self:GetTargets()

local AutoFarmSuper = {
    Active = false,
    Mode = "Kill", -- Kill, Click, Collect, Hybrid
    Delay = 0.1,
    FOV = 200,
    MaxDistance = 1000,
    TeamCheck = true,
    VisibleCheck = true,
    TargetPart = "Head",
    Prediction = true,
    PredictionAmount = 0.2,
    Silent = false,
    OnlyHead = true,
    MinHealth = 0,
    AutoSwitch = true,
    SwitchDelay = 2,
    MultiTarget = false,
    TargetLimit = 3,
    BypassCooldown = false,
    Stats = {Kills = 0, Clicks = 0, Collects = 0, StartTime = 0, Coins = 0},
    CurrentTarget = nil,
    Targets = {},
}

function AutoFarmSuper:FindTargets()
    local targets = {}
    local center = Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and Utils.IsAlive(plr) then
            if self.TeamCheck and Utils.IsTeamMate(plr) then continue end
            local dist = Utils.GetDistance(plr)
            if dist > self.MaxDistance then continue end
            if self.VisibleCheck and not Utils.IsVisible(plr) then continue end
            
            local part = self.OnlyHead and Utils.GetPart(plr, "Head") or Utils.GetPart(plr, "HumanoidRootPart")
            if not part then continue end
            
            local pos, onScreen = Camera:WorldToScreenPoint(part.Position)
            if not onScreen then continue end
            
            local fovDist = (Vector2.new(pos.X, pos.Y) - center).magnitude
            if fovDist > self.FOV then continue end
            
            local hum = Utils.GetHumanoid(plr)
            if hum and hum.Health <= self.MinHealth then continue end
            
            table.insert(targets, {
                player = plr,
                distance = dist,
                fov = fovDist,
                health = hum and hum.Health or 0
            })
        end
    end
    
    -- Sort by distance
    table.sort(targets, function(a, b) return a.distance < b.distance end)
    return targets
end

function AutoFarmSuper:GetBestTarget()
    local targets = self:FindTargets()
    if #targets > 0 then
        if self.MultiTarget then
            local limit = math.min(self.TargetLimit, #targets)
            self.Targets = {}
            for i = 1, limit do
                table.insert(self.Targets, targets[i].player)
            end
            return self.Targets[1]
        else
            self.Targets = {targets[1].player}
            return targets[1].player
        end
    end
    return nil
end

function AutoFarmSuper:Shoot(target)
    if not target then return false end
    if self.BypassCooldown then
        local currentTime = os.time()
        if self.LastShootTime and currentTime - self.LastShootTime < 0.05 then
            return false
        end
        self.LastShootTime = currentTime
    end
    
    local remote = Trigger.ShootRemote or Trigger:FindShootRemote()
    if remote then
        remote:FireServer()
        self.Stats.Clicks = self.Stats.Clicks + 1
        return true
    else
        VirtualUser:CaptureController()
        VirtualUser:ClickButton(Vector2.new(0, 0))
        self.Stats.Clicks = self.Stats.Clicks + 1
        return true
    end
end

function AutoFarmSuper:ShootTargets()
    if self.MultiTarget then
        for _, target in pairs(self.Targets) do
            if target and Utils.IsAlive(target) then
                self:Shoot(target)
            end
        end
    else
        local target = self.CurrentTarget
        if target and Utils.IsAlive(target) then
            self:Shoot(target)
            local hum = Utils.GetHumanoid(target)
            if hum and hum.Health <= 0 then
                self.Stats.Kills = self.Stats.Kills + 1
                self.CurrentTarget = nil
            end
        end
    end
end

            self:CollectItems()

            self:AutoSwitchTarget()
        else
            if not self.CurrentTarget or not Utils.IsAlive(self.CurrentTarget) then
                self.CurrentTarget = self:GetBestTarget()
            end
        end
        
        -- Shoot
        if self.Mode == "Kill" or self.Mode == "Click" or self.Mode == "Hybrid" then
            if self.MultiTarget then
                self:ShootTargets()
            else
                if self.CurrentTarget and Utils.IsAlive(self.CurrentTarget) then

                    local hum = Utils.GetHumanoid(self.CurrentTarget)
                    if hum and hum.Health <= 0 then
                        self.Stats.Kills = self.Stats.Kills + 1
                        self.CurrentTarget = nil
                    end
                end
            end
        end
        
        wait(self.Delay)
    end
end

function Radar:UpdateDot(plr, frame)
    if not plr or not plr.Character then return end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    local selfRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not selfRoot then return end
    
    local diff = root.Position - selfRoot.Position
    local dist = diff.magnitude
    if dist > self.Range then
        if frame then frame.Visible = false end
        return
    end
    
    local angle = math.atan2(diff.X, diff.Z)
    local rad = (dist / self.Range) * (self.Size / 2)
    
    local dot = frame or Instance.new("Frame")
    dot.Size = UDim2.new(0, 5, 0, 5)
    dot.BackgroundColor3 = Utils.IsTeamMate(plr) and self.ColorTeammate or self.ColorEnemy
    dot.BackgroundTransparency = 0
    dot.BorderSizePixel = 0
    dot.Visible = true
    dot.Parent = self.Frame
    dot.Position = UDim2.new(0.5, -2.5 + math.sin(angle) * rad, 0.5, -2.5 + math.cos(angle) * rad)
    
    -- Name label
    if self.ShowNames then
        local label = dot:FindFirstChild("Name") or Instance.new("TextLabel")
        label.Size = UDim2.new(0, 50, 0, 15)
        label.Position = UDim2.new(0, -22.5, 0, 5)

        label.TextColor3 = Color3.fromRGB(255, 255, 255)
        label.BackgroundTransparency = 1
        label.TextScaled = true
        label.Font = Enum.Font.Gotham
        label.Parent = dot
    end
    
    -- Distance
    if self.ShowDistance then
        local distLabel = dot:FindFirstChild("Dist") or Instance.new("TextLabel")
        distLabel.Size = UDim2.new(0, 30, 0, 12)
        distLabel.Position = UDim2.new(0, -10, 1, 0)

        distLabel.BackgroundTransparency = 1
        distLabel.TextScaled = true
        distLabel.Font = Enum.Font.Gotham
        distLabel.Parent = dot
    end
    
    if not frame then
        table.insert(self.Dots, dot)
    end
end

function Radar:Update()
    while self.Active do
        -- Cleanup dead dots
        for i, dot in pairs(self.Dots) do
            if not dot or not dot.Parent then
                table.remove(self.Dots, i)
            end
        end
        
        -- Update dots
        local index = 1
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer then
                local dot = self.Dots[index]
                if dot then
                    self:UpdateDot(plr, dot)
                else
                    self:UpdateDot(plr, nil)
                end
                index = index + 1
            end
        end
        
        -- Hide remaining dots
        for i = index, #self.Dots do
            self.Dots[i].Visible = false
        end
        
        wait(0.1)
    end
end

function MusicPlayer:Play(index)
    if #self.Playlist == 0 then return end
    if self.CurrentSong then
        self.CurrentSong:Stop()
        self.CurrentSong:Destroy()
    end
    
    index = index or self.CurrentIndex
    if index > #self.Playlist then index = 1 end
    if index < 1 then index = #self.Playlist end
    
    local song = self.Playlist[index]
    if not song then return end
    
    self.CurrentIndex = index
    self.CurrentSong = SoundSystem:Play(song.id, self.Volume, workspace, true)
    print("Now playing:", song.name)
end

function MusicPlayer:Next()
    local index = self.CurrentIndex + 1
    if index > #self.Playlist then
        if self.Loop then index = 1
        else return end
    end
    self:Play(index)
end

function MusicPlayer:Previous()
    local index = self.CurrentIndex - 1
    if index < 1 then
        if self.Loop then index = #self.Playlist
        else return end
    end
    self:Play(index)
end

function MusicPlayer:Stop()
    if self.CurrentSong then
        self.CurrentSong:Stop()
        self.CurrentSong:Destroy()
        self.CurrentSong = nil
    end
    print("Music stopped")
end

    if self.Active then
        if #self.Playlist > 0 then self:Play() end

    if self.Stats.Streak > self.Stats.BestStreak then
        self.Stats.BestStreak = self.Stats.Streak
    end
    if headshot then self.Stats.Headshots = self.Stats.Headshots + 1 end
    print("💀 Kill! Total:", self.Stats.Kills, "Streak:", self.Stats.Streak)
end

function Statistics:AddDeath()
    self.Stats.Deaths = self.Stats.Deaths + 1
    self.Stats.Streak = 0
    print("💀 Death! Total:", self.Stats.Deaths)
end

end

    if self.Stats.Deaths == 0 then return self.Stats.Kills end

local ColorUtils = {
    Active = true,
}

function ColorUtils:HexToRGB(hex)
    hex = hex:gsub("#", "")
    return {
        r = tonumber(hex:sub(1, 2), 16),
        g = tonumber(hex:sub(3, 4), 16),
        b = tonumber(hex:sub(5, 6), 16),
    }
end

function ColorUtils:RGBToHex(r, g, b)
    return string.format("#%02x%02x%02x", r, g, b)
end

function ColorUtils:Color3ToHex(color)
    return string.format("#%02x%02x%02x", color.R * 255, color.G * 255, color.B * 255)
end

function ColorUtils:HexToColor3(hex)
    local rgb = self:HexToRGB(hex)
    return Color3.new(rgb.r / 255, rgb.g / 255, rgb.b / 255)
end

function ColorUtils:ColorLerp(c1, c2, t)
    return Color3.new(
        c1.R + (c2.R - c1.R) * t,
        c1.G + (c2.G - c1.G) * t,
        c1.B + (c2.B - c1.B) * t
    )
end

                obj.gui:Destroy()

            end
        end

        for _, point in pairs(self.Points) do
            local exists = false
            for _, obj in pairs(self.Objects) do
                if obj.part.Position == point.position then
                    exists = true; break
                end
            end
            if not exists then
                self:CreateArrow(point.position)
            end
        end
        wait(1)
    end

        return TeleportService:GetServerList()
    end)
    if success and data then
        for _, server in pairs(data) do
            table.insert(servers, server)
        end
    end
    return servers
end

function AutoReconnect:FindBestServer()
    local servers = self:GetServers()
    if #servers == 0 then
        -- Gunakan server default
        return nil
    end
    -- Cari server dengan ping terendah
    table.sort(servers, function(a, b)
        return (a.Ping or 999) < (b.Ping or 999)
    end)
    -- Pilih server pertama atau preferred
    if self.PreferredServer then
        for _, server in pairs(servers) do
            if server.Name == self.PreferredServer then
                return server
            end
        end
    end
    return servers[1]
end

function AutoReconnect:Reconnect()
    if self.Attempts >= self.MaxAttempts then
        print("❌ Max reconnect attempts reached")
        return false
    end
    
    self.Attempts = self.Attempts + 1
    print("🔄 Reconnect attempt " .. self.Attempts .. "/" .. self.MaxAttempts)
    
    local server = self:FindBestServer()
    if server then
        local success = pcall(function()
            TeleportService:Teleport(server)
        end)
        if success then
            self.Attempts = 0
            print("✅ Reconnected successfully")
            if self.OnReconnect then
                self:OnReconnect()
            end
            return true
        end
    else
        -- Alternative: reload game
        local success = pcall(function()
            TeleportService:Teleport(game.PlaceId)
        end)
        if success then
            self.Attempts = 0
            print("✅ Reconnected (reload)")
            return true
        end
    end
    
    print("❌ Reconnect attempt failed")
    return false
end

        self.ProxyIndex = 1
    end
    return proxy
end

    if not proxy then return false end
    self.CurrentProxy = proxy

    local success = pcall(function()

    if self.TestProxy then

            print("✅ Proxy test passed")

local Logger = {
    Active = false,
    LogFile = "ROFF_Log.txt",
    LogBuffer = {},

    LogLevels = {
        INFO = 1,
        WARNING = 2,
        ERROR = 3,
        DEBUG = 4,
        SUCCESS = 5,
        KILL = 6,
        DAMAGE = 7,
    },
    CurrentLevel = 1,
    AutoFlush = true,
    FlushInterval = 60,
    LastFlush = 0,
    IncludeTimestamp = true,
    IncludeSource = true,

function Logger:AddLog(level, message, source)
    if level < self.CurrentLevel then return end
    if not self.Active then return end
    
    local time = os.date("%Y-%m-%d %H:%M:%S")
    local levelName = ""
    for k, v in pairs(self.LogLevels) do
        if v == level then
            levelName = k
            break
        end
    end
    
    local logEntry = ""
    if self.IncludeTimestamp then
        logEntry = logEntry .. "[" .. time .. "]"
    end
    if self.IncludeSource and source then
        logEntry = logEntry .. "[" .. source .. "]"
    end
    logEntry = logEntry .. "[" .. levelName .. "] " .. message
    
    table.insert(self.LogBuffer, logEntry)
    if #self.LogBuffer > self.MaxBufferSize then
        self:Flush()
    end
end

function Logger:Flush()
    if #self.LogBuffer == 0 then return end
    local content = table.concat(self.LogBuffer, "\n") .. "\n"
    -- Simulasi write file
    -- Untuk implementasi nyata: writefile(self.LogFile, content)
    print("📝 Log flushed (" .. #self.LogBuffer .. " entries)")
    self.LogBuffer = {}
    self.LastFlush = os.time()
end

    self:AddLog(self.LogLevels.INFO, message, source)
end

    self:AddLog(self.LogLevels.WARNING, message, source)
end

function Logger:Error(message, source)

function AuctionSystem:PlaceBid(auctionId, amount, bidder)
    local auction = self.Auctions[auctionId]
    if not auction then
        print("❌ Auction not found")
        return false
    end
    
    if auction.status ~= "active" then
        print("❌ Auction is not active")
        return false
    end
    
    if amount < auction.currentBid + self.MinBidIncrement then
        print("❌ Bid must be at least:", auction.currentBid + self.MinBidIncrement)
        return false
    end
    
    auction.currentBid = amount
    auction.highestBidder = bidder or LocalPlayer
    table.insert(auction.bids, {bidder = bidder or LocalPlayer, amount = amount, time = os.time()})
    print("💰 New bid placed:", amount, "by", (bidder or LocalPlayer).Name)
    Logger:Info("Bid placed on auction " .. auctionId .. ": " .. amount, "AuctionSystem")
    return true
end

function AuctionSystem:Buyout(auctionId, buyer)

    if not auction then

    if not auction.buyout then
        print("❌ No buyout price set")
        return false
    end
    
    if auction.status ~= "active" then
        print("❌ Auction is not active")
        return false
    end
    
    auction.highestBidder = buyer or LocalPlayer
    auction.currentBid = auction.buyout
    auction.status = "completed"
    print("💎 Auction bought out for:", auction.buyout)
    Logger:Info("Auction " .. auctionId .. " bought out", "AuctionSystem")

function MacroSystem:PlayMacro()
    if self.IsPlaying then
        print("⚠️ Already playing")
        return false
    end
    if #self.RecordedActions == 0 then
        print("⚠️ No actions recorded")
        return false
    end
    
    self.IsPlaying = true
    self.CurrentAction = 1
    print("▶️ Playing macro...")
    
    task.spawn(function()
        while self.IsPlaying and self.CurrentAction <= #self.RecordedActions do
            local action = self.RecordedActions[self.CurrentAction]
            self:ExecuteAction(action)
            self.CurrentAction = self.CurrentAction + 1
            wait(self.PlaybackSpeed * 0.01)
        end
        
        if self.LoopPlayback and self.IsPlaying then
            self.CurrentAction = 1
            self:PlayMacro()
        else
            self.IsPlaying = false
            print("⏹️ Macro playback finished")
        end
    end)
    
    return true
end

function MacroSystem:ExecuteAction(action)
    if action.type == "click" then
        local data = action.data
        VirtualUser:CaptureController()
        VirtualUser:ClickButton(Vector2.new(data.x or 0, data.y or 0))
    elseif action.type == "keypress" then
        -- Simulasi keypress
        local key = action.data.key
        VirtualUser:CaptureController()
        VirtualUser:KeyDown(key)
        wait(0.05)
        VirtualUser:KeyUp(key)
    elseif action.type == "delay" then
        wait(action.data.ms / 1000)
    elseif action.type == "mousemove" then
        local data = action.data
        UserInputService.MousePosition = Vector2.new(data.x or 0, data.y or 0)
    end
end

local ScriptExecutor = {

function ScriptExecutor:AddScript(name, code, author)

        name = name,
        code = code,
        author = author or "Unknown",
        timestamp = os.time(),
    })
    print("✅ Script added:", name)
    return true
end

function ScriptExecutor:LoadScript(name)
    for _, script in pairs(self.Scripts) do
        if script.name == name then
            self.CurrentScript = script
            table.insert(self.ScriptHistory, script)
            if #self.ScriptHistory > self.MaxHistory then
                table.remove(self.ScriptHistory, 1)
            end
            print("📂 Script loaded:", name)
            return script
        end
    end
    print("❌ Script not found:", name)
    return nil
end

function ScriptExecutor:ExecuteScript(script)
    if not script then
        if self.CurrentScript then
            script = self.CurrentScript
        else
            print("❌ No script to execute")
            return false
        end
    end
    
    local success, err = pcall(function()
        loadstring(script.code)()
    end)
    
    if success then
        print("✅ Script executed:", script.name)
        Logger:Info("Script executed: " .. script.name, "ScriptExecutor")
        return true
    else
        print("❌ Script execution failed:", err)
        Logger:Error("Script execution failed: " .. err, "ScriptExecutor")
        return false
    end
end

function ScriptExecutor:ExecuteByName(name)
    local script = self:LoadScript(name)
    if script then
        return self:ExecuteScript(script)
    end
    return false
end

function ScriptExecutor:ExecuteFromURL(url)
    local success, code = pcall(function()
        return game:HttpGet(url)
    end)
    
    if success and code then
        local script = {
            name = "URL_" .. os.time(),
            code = code,
            author = "URL",
            timestamp = os.time(),
        }
        table.insert(self.Scripts, script)
        return self:ExecuteScript(script)
    else
        print("❌ Failed to load script from URL:", url)
        return false
    end
end

function ScriptExecutor:AutoRunScript(name)
    self.AutoRun = true
    self.AutoRunScriptName = name

        while self.AutoRun do
            self:ExecuteByName(name)
            wait(self.AutoRunDelay)
        end
    end)
    print("🔄 Auto-run started for:", name)
end

    self.AutoRun = false
    print("⏹️ Auto-run stopped")
end
