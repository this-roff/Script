local Utils = {}

function Utils.GetClosestPlayer(fov)
    fov = fov or 9999
    local closest, distance = nil, math.huge
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local root = plr.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
                if dist < distance and dist < fov then
                    closest, distance = plr, dist
                end
            end
        end
    end
    return closest
end

function Utils.GetPlayersInFOV(fov)
    local players = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
            local root = plr.Character.HumanoidRootPart
            local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
            if onScreen then
                local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
                if dist < fov then
                    table.insert(players, {player = plr, distance = dist})
                end
            end
        end
    end
    table.sort(players, function(a, b) return a.distance < b.distance end)
    return players
end

function Utils.IsTeamMate(plr)
    if not Config.AimTeamCheck then return false end
    return plr.Team == LocalPlayer.Team
end

function Utils.GetPart(plr, partName)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild(partName) or plr.Character:FindFirstChild("HumanoidRootPart")
end

function Utils.GetHumanoid(plr)
    if not plr or not plr.Character then return nil end
    return plr.Character:FindFirstChild("Humanoid")
end

function Utils.IsAlive(plr)
    local hum = Utils.GetHumanoid(plr)
    return hum and hum.Health > 0
end

function Utils.GetDistance(plr)
    if not plr or not plr.Character then return math.huge end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return math.huge end
    return (Camera.CFrame.Position - root.Position).magnitude
end

function Utils.IsVisible(plr)
    if not plr or not plr.Character then return false end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local ray = Ray.new(Camera.CFrame.Position, (root.Position - Camera.CFrame.Position).unit * 500)
    local hit, position = Workspace:FindPartOnRay(ray, LocalPlayer.Character)
    if hit then
        local hitPlr = hit.Parent and hit.Parent.Parent and hit.Parent.Parent:FindFirstChild("Humanoid")
        if hitPlr then
            return hitPlr == Utils.GetHumanoid(plr)
        end
    end
    return false
end

function Utils.GetPlayerColor(plr)
    if Config.ESPTeamColor and Utils.IsTeamMate(plr) then
        return Config.ESPTeamColorTeammate
    else
        return Config.ESPTeamColorEnemy
    end
end

function Utils.RandomColor()
    return Color3.fromRGB(math.random(0, 255), math.random(0, 255), math.random(0, 255))
end

function Utils.Lerp(a, b, t)
    return a + (b - a) * t
end

function Utils.Clamp(value, min, max)
    return math.max(min, math.min(max, value))
end

function Utils.SmoothDamp(current, target, currentVelocity, smoothTime, maxSpeed, deltaTime)
    smoothTime = math.max(0.0001, smoothTime)
    local omega = 2.0 / smoothTime
    local x = omega * deltaTime
    local exp = 1 / (1 + x + 0.48 * x * x + 0.235 * x * x * x)
    local change = current - target
    local maxChange = maxSpeed * smoothTime
    change = math.min(maxChange, math.max(-maxChange, change))
    target = current - change
    local temp = (currentVelocity + omega * change) * deltaTime
    currentVelocity = (currentVelocity - omega * temp) * exp
    local output = target + (change + temp) * exp
    if output > 1 or output < 0 then
        output = target
        currentVelocity = 0
    end
    return output, currentVelocity
end

local Aim = {
    LockedTarget = nil,
    Smoothness = 0.25,
    FOV = 150,
    TargetPart = "Head",
    Prediction = false,
    PredictionAmount = 0.2,
    SilentAim = false,
    Active = false,
    SilentActive = false,
    CurrentTarget = nil,
}

function Aim:FindTarget()
    local target = Utils.GetClosestPlayer(self.FOV)
    if target and Utils.IsAlive(target) then
        return target
    end
    return nil
end

function Aim:GetTargetPosition(plr)
    local part = Utils.GetPart(plr, self.TargetPart)
    if not part then return nil end
    local pos = part.Position
    if self.Prediction then
        local velocity = plr.Character.HumanoidRootPart.Velocity
        pos = pos + velocity * self.PredictionAmount
    end
    return pos
end

function Aim:Lock()
    while self.Active do
        local target = self:FindTarget()
        if target then
            self.LockedTarget = target
            local pos = self:GetTargetPosition(target)
            if pos then
                local lookAt = CFrame.lookAt(Camera.CFrame.Position, pos)
                Camera.CFrame = Camera.CFrame:Lerp(lookAt, self.Smoothness)
            end
        end
        wait(0.01)
    end
end

function Aim:SilentAim()
    while self.SilentActive do
        local target = self:FindTarget()
        if target then
            local pos = self:GetTargetPosition(target)
            if pos then
                local lookAt = CFrame.lookAt(Camera.CFrame.Position, pos)
                Camera.CFrame = lookAt
            end
        end
        wait(0.01)
    end
end

function Aim:Assist()
    while Config.AimAssist do
        local target = self:FindTarget()
        if target then
            local pos = self:GetTargetPosition(target)
            if pos then
                local lookAt = CFrame.lookAt(Camera.CFrame.Position, pos)
                Camera.CFrame = Camera.CFrame:Lerp(lookAt, Config.AimAssistStrength)

        wait(0.01)
    end
end

function Aim:Toggle()

local function CreateESP(plr)
    if not plr.Character then return end
    local root = plr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return end
    
    local espData = {}
    local color = GetPlayerColor(plr)
    
    -- Update warna berdasarkan team
    if ESP.Box then
        local box = CreateESPBox(plr)
        if box then
            box.Color3 = color
            espData.box = box
        end
    end
    
    if ESP.Name then
        local name = CreateESPName(plr)
        if name then
            name.TextLabel.TextColor3 = color
            espData.name = name
        end
    end
    
    if ESP.Line then
        local line = CreateESPLine(plr)
        if line then
            line.Color3 = color
            espData.line = line
        end
    end
    
    if ESP.Health then
        local health = CreateESPHealth(plr)
        if health then
            espData.health = health
        end
    end
    
    if ESP.Distance then
        local dist = CreateESPDistance(plr)
        if dist then
            espData.distance = dist
        end
    end
    
    if ESP.Chams then
        local chams = CreateESPChams(plr)
        if chams then
            espData.chams = chams
        end
    end
    
    espData.plr = plr
    table.insert(ESP.Objects, espData)
end

local function UpdateESP()
    while ESP.Active do
        for i, esp in pairs(ESP.Objects) do
            if not esp.plr or not esp.plr.Character then
                if esp.box then esp.box:Destroy() end
                if esp.name then esp.name:Destroy() end
                if esp.line then esp.line:Destroy() end
                if esp.health then esp.health:Destroy() end
                if esp.distance then esp.distance:Destroy() end
                if esp.chams then
                    for _, data in pairs(esp.chams) do
                        if data.part then
                            data.part.Color = data.color
                            data.part.Material = data.material
                        end
                    end
                end
                table.remove(ESP.Objects, i)
            else
                -- Update distance
                if esp.distance then
                    local dist = Utils.GetDistance(esp.plr)
                    esp.distance.Text = math.floor(dist) .. "m"
                end
                -- Update health
                if esp.health then
                    local humanoid = Utils.GetHumanoid(esp.plr)
                    if humanoid then
                        local healthPercent = humanoid.Health / humanoid.MaxHealth
                        esp.health.Frame.Size = UDim2.new(healthPercent, 0, 1, 0)
                    end
                end
                -- Update color berdasarkan team
                local color = GetPlayerColor(esp.plr)
                if esp.box then esp.box.Color3 = color end
                if esp.name then esp.name.TextLabel.TextColor3 = color end
                if esp.line then esp.line.Color3 = color end
            end
        end
        
        -- Create ESP for new players
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local exists = false
                for _, esp in pairs(ESP.Objects) do
                    if esp.plr == plr then exists = true; break end
                end
                if not exists then
                    CreateESP(plr)
                end
            end
        end
        wait(0.1)
    end
end

local function ToggleESP()
    ESP.Active = not ESP.Active
    if ESP.Active then
        task.spawn(UpdateESP)
        print("ESP ON")
    else
        for _, esp in pairs(ESP.Objects) do
            if esp.box then esp.box:Destroy() end
            if esp.name then esp.name:Destroy() end

    local root = Utils.GetPart(plr, "HumanoidRootPart")
    if not root then return nil end
    local line = Instance.new("LineHandleAdornment")
    line.Length = 50
    line.Thickness = Tracers.Thickness

    return line

local function UpdateTracers()
    while Tracers.Active do
        for i, tracer in pairs(Tracers.Objects) do
            if not tracer.plr or not tracer.plr.Character then
                tracer.line:Destroy()
                table.remove(Tracers.Objects, i)
            end
        end
        
        for _, plr in pairs(Players:GetPlayers()) do
            if plr ~= LocalPlayer and plr.Character then
                local exists = false
                for _, tracer in pairs(Tracers.Objects) do
                    if tracer.plr == plr then exists = true; break end
                end
                if not exists then
                    local line = CreateTracer(plr)
                    if line then
                        table.insert(Tracers.Objects, {plr = plr, line = line})
                    end
                end
            end
        end
        wait(0.5)
    end
end

end

local ErrorHandler = {
    Errors = {},
    MaxErrors = 10,
    RecoveryAttempts = 0,
    AutoRecover = true,
}

function ErrorHandler:HandleError(err, source)
    local errorInfo = {
        time = os.time(),
        source = source or "Unknown",
        message = tostring(err),
    }
    table.insert(self.Errors, errorInfo)
    if #self.Errors > self.MaxErrors then
        table.remove(self.Errors, 1)
    end
    print("⚠️ Error caught: " .. tostring(err))
    if self.AutoRecover then
        self:AttemptRecovery()
    end
end

function ErrorHandler:AttemptRecovery()
    self.RecoveryAttempts = self.RecoveryAttempts + 1
    if self.RecoveryAttempts > 3 then
        print("❌ Too many recovery attempts, stopping...")
        return
    end
    print("🔄 Attempting recovery... (" .. self.RecoveryAttempts .. ")")
    -- Reset beberapa sistem
    if Aim then
        Aim.Active = false
        Aim.SilentActive = false
    end
    if Trigger then
        Trigger.Active = false
    end
    if ESP then
        ESP.Active = false
    end
    wait(2)
    print("✅ Recovery completed")
end

function ErrorHandler:Start()
    -- Pasang error handler global
    local oldHandler = xpcall
    xpcall = function(f, errHandler)
        return oldHandler(f, function(err)
            self:HandleError(err, "Global")
            if errHandler then
                return errHandler(err)
            end
        end)
    end
    print("✅ Error Handler System Started")
end

    PredictionAmount = 0.2,

    SwitchDelay = 2,
    CurrentTarget = nil,
    LastTarget = nil,
    SwitchTimer = 0,
    MinHealth = 0,
    MaxDistance = 500,
}

function AdvancedAim:FindBestTarget()
    local candidates = {}
    for _, plr in pairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and Utils.IsAlive(plr) then
            local distance = Utils.GetDistance(plr)
            if distance < self.MaxDistance then
                local health = Utils.GetHumanoid(plr).Health
                local score = 0
                if self.Mode == "FOV" then
                    local root = Utils.GetPart(plr, "HumanoidRootPart")
                    if root then
                        local pos, onScreen = Camera:WorldToScreenPoint(root.Position)
                        if onScreen then
                            local dist = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).magnitude
                            score = 1000 - dist
                        end
                    end
                elseif self.Mode == "Distance" then
                    score = 1000 - distance
                elseif self.Mode == "Health" then
                    score = health
                end
                table.insert(candidates, {player = plr, score = score})
            end
        end
    end
    table.sort(candidates, function(a, b) return a.score > b.score end)
    if #candidates > 0 then
        return candidates[1].player
    end
    return nil
end

function AdvancedAim:GetTargetPosition(plr)
    local part = Utils.GetPart(plr, self.TargetPart)
    if not part then return nil end
    local pos = part.Position
    if self.Prediction then
        local velocity = plr.Character.HumanoidRootPart.Velocity
        pos = pos + velocity * self.PredictionAmount
    end
    return pos
end

function AdvancedAim:Run()
    while self.Active do
        local target = self:FindBestTarget()
        if target then
            self.CurrentTarget = target
            local pos = self:GetTargetPosition(target)
            if pos then
                local lookAt = CFrame.lookAt(Camera.CFrame.Position, pos)
                Camera.CFrame = Camera.CFrame:Lerp(lookAt, self.Smoothness)
            end
        end
        wait(0.01)
    end
end

function AutoReconnect:GetServers()
    local servers = {}
    -- Dapatkan daftar server dari TeleportService
    local success, data = pcall(function()
        return TeleportService:GetServerList()
    end)
    if success then
        for _, server in pairs(data) do
            table.insert(servers, server)
        end
    end
    return servers
end

function AutoReconnect:FindBestServer()
    local servers = self:GetServers()
    if #servers == 0 then
        -- Gunakan server default
        return nil
    end
    -- Cari server dengan ping terendah
    table.sort(servers, function(a, b)
        return a.Ping < b.Ping
    end)
    return servers[1]
end

function AutoReconnect:Reconnect()
    local server = self:FindBestServer()
    if server then
        print("🔄 Attempting to reconnect to server: " .. server.Name)
        local success = pcall(function()
            TeleportService:Teleport(server)
        end)
        if success then
            self.Attempts = 0
            print("✅ Reconnected successfully")
        else
            self.Attempts = self.Attempts + 1
            print("❌ Reconnect attempt failed: " .. self.Attempts)
        end
    else
        print("❌ No server available")
    end
end

function Security:VerifyChecksum()

function Security:AntiTamper()
    while self.AntiTamper do
        if not self:VerifyChecksum() then
            print("⚠️ Tampering detected! Restoring...")
            -- Restore config
            for k, v in pairs(Config) do
                Config[k] = v
            end
        end
        wait(5)
    end
end

function Security:AntiDebug()
    while self.AntiDebug do
        -- Detect debugger
        local success = pcall(function()
            return debug.getinfo(1)
        end)
        if not success then

            -- Emergency shutdown
            self:EmergencyShutdown()

        wait(10)
    end
end

function Security:AddBlacklist(plr)

end

function Security:IsWhitelisted(plr)
    for _, name in pairs(self.Whitelist) do
        if name == plr.Name then return true end
    end
    return false
end

function Security:IsBlacklisted(plr)
    for _, name in pairs(self.Blacklist) do
        if name == plr.Name then return true end
    end
    return false

-- Fungsi drag
function LogoSystem:SetupDrag(frame)
    local dragging = false
    local dragInput = nil
    local dragStart = nil
    local startPos = nil

    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            dragInput = input
        end
    end)

    frame.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    game:GetService("RunService").RenderStepped:Connect(function()
        if dragging and dragInput and startPos then
            local delta = dragInput.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale + delta.X / game:GetService("GuiService").ScreenSize.X,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale + delta.Y / game:GetService("GuiService").ScreenSize.Y,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

function LogoSystem:MinimizeGUI()
    local gui = LocalPlayer.PlayerGui:FindFirstChild("FPS_DEWA")
    if gui then
        local mainFrame = gui:FindFirstChildOfClass("Frame")
        if mainFrame then
            -- Animasi mengecil
            local tween = TweenService:Create(mainFrame, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                Size = UDim2.new(0, 0, 0, 0)
            })
            tween:Play()
            tween.Completed:Wait()
            gui:Destroy()
            self.IsMinimized = true
            self:ShowLogo()
        end
    end
end

function LogoSystem:ShowLogo()
    -- Tampilkan logo dengan animasi
    if self.LogoButton then
        self.LogoButton.Size = UDim2.new(0, 0, 0, 0)
        local tween = TweenService:Create(self.LogoButton, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, self.LogoSize, 0, self.LogoSize)
        })
        tween:Play()
    end
end

function LogoSystem:RestoreGUI()
    self.IsMinimized = false
    -- Sembunyikan logo
    self:AnimateLogo(self.LogoButton, false)
    wait(0.3)
    self:CreateMainGUI()
end

-- ============================================================
-- MODULE 56: START LOGO SYSTEM
-- ============================================================
function LogoSystem:Start()
    self:CreateLogo()
    print("✅ Logo System Started")
end

-- Jalankan logo system
LogoSystem:Start()
print("✅ PART 11 LOADED - LOGO & ANIMATION SYSTEM (UPDATED WITH 15-DIGIT ID)")
